   284→// Live Clock
   285→// ──────────────────────────────────────────────────────
   286→function LiveClock() {
   287→  const [time, setTime] = React.useState("")
   288→
   289→  React.useEffect(() => {
   290→    const tick = () => {
   291→      setTime(
   292→        new Date().toLocaleTimeString("en-IN", {
   293→          hour: "2-digit",
   294→          minute: "2-digit",
   295→          second: "2-digit",
   296→          hour12: true,
   297→        })
   298→      )
   299→    }
   300→    tick()
   301→    const id = setInterval(tick, 1000)
   302→    return () => clearInterval(id)
   303→  }, [])
   304→
   305→  if (!time) return null
   306→  return (
   307→    <span className="hidden items-center gap-1.5 text-xs text-muted-foreground lg:flex">
   308→      <Clock className="h-3 w-3" />
   309→      {time}
   310→    </span>
   311→  )
   312→}

   526→      if (e.key === "k" && (e.metaKey || e.ctrlKey)) {
   527→        e.preventDefault()
   528→        setCommandPaletteOpen(!commandPaletteOpen)
   529→      }
   530→      if (e.key === "Escape" && commandPaletteOpen) {
   531→        setCommandPaletteOpen(false)
   532→        setSearch("")
   533→      }
   534→    }
   535→    document.addEventListener("keydown", down)
   536→    return () => document.removeEventListener("keydown", down)
   537→  }, [commandPaletteOpen, setCommandPaletteOpen])
   538→
   539→  React.useEffect(() => {
   540→    if (commandPaletteOpen && inputRef.current) {
   541→      inputRef.current.focus()
   542→    }
   543→  }, [commandPaletteOpen])
   544→
   545→  const filtered = navItems.filter((item) =>
   546→    item.label.toLowerCase().includes(search.toLowerCase())
   547→  )
   548→
   549→  return (
   550→    <>
   551→      <div
   552→        className={cn(
   553→          "fixed inset-0 z-50 bg-black/40 backdrop-blur-sm transition-opacity duration-200",
   554→          commandPaletteOpen ? "opacity-100" : "pointer-events-none opacity-0"
   555→        )}
   556→        onClick={() => { setCommandPaletteOpen(false); setSearch("") }}
   557→      />
   558→      <div
   559→        className={cn(
   560→          "fixed left-1/2 top-[15%] z-50 w-full max-w-lg -translate-x-1/2 rounded-2xl border border-border/60 bg-background shadow-2xl shadow-black/20 transition-all duration-200",
   561→          commandPaletteOpen ? "scale-100 opacity-100" : "scale-95 opacity-0 pointer-events-none"
   562→        )}
   563→      >
   564→        <div className="flex items-center gap-2 border-b px-4">
   565→          <Search className="h-4 w-4 shrink-0 text-muted-foreground" />
   566→          <input
   567→            ref={inputRef}
   568→            placeholder="Search pages, warehouses, actions..."
   569→            className="flex-1 bg-transparent py-3.5 text-sm outline-none placeholder:text-muted-foreground"
   570→            value={search}
   571→            onChange={(e) => setSearch(e.target.value)}
   572→          />
   573→          <kbd className="pointer-events-none inline-flex h-5 select-none items-center rounded border bg-muted/80 px-1.5 font-mono text-[10px] font-medium text-muted-foreground">
   574→            ESC
   575→          </kbd>
   576→        </div>
   577→        <div className="max-h-72 overflow-y-auto p-2">
   578→          <div className="px-2 py-1.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   579→            {search ? "Results" : "Quick Navigation"}
   580→          </div>
   581→          {filtered.map((item) => {
   582→            const Icon = iconMap[item.icon]
   583→            const isActive = useAppStore.getState().activeView === item.id
   584→            return (
   585→              <button
   586→                key={item.id}
   587→                className={cn(
   588→                  "flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm text-left transition-colors",
   589→                  isActive ? "bg-blue-50 text-blue-700 dark:bg-blue-950 dark:text-blue-300" : "hover:bg-muted"
   590→                )}
   591→                onClick={() => {
   592→                  setActiveView(item.id)
   593→                  setCommandPaletteOpen(false)
   594→                  setSearch("")
   595→                }}
   596→              >
   597→                {Icon ? <Icon className="h-4 w-4 shrink-0" /> : <div className="h-4 w-4" />}
   598→                <span className="flex-1">{item.label}</span>
   599→                {isActive && <span className="text-[10px] text-muted-foreground">Current</span>}
   600→              </button>
   601→            )
   602→          })}
   603→          {filtered.length === 0 && (
   604→            <div className="px-3 py-8 text-center text-sm text-muted-foreground">
   605→              No results for &ldquo;{search}&rdquo;
   606→            </div>
   607→          )}
   608→        </div>
   609→        <div className="border-t px-4 py-2 flex items-center justify-between text-[10px] text-muted-foreground">
   610→          <span>Type to search</span>
   611→          <span className="flex items-center gap-1">
   612→            <kbd className="rounded border bg-muted px-1 py-0.5 font-mono text-[9px]">↑↓</kbd> navigate
   613→            <kbd className="rounded border bg-muted px-1 py-0.5 font-mono text-[9px] ml-2">↵</kbd> select
   614→          </span>
   615→        </div>
   616→      </div>
   617→    </>
   618→  )
   619→}
