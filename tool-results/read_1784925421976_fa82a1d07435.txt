   286→// Live Clock
   287→// ──────────────────────────────────────────────────────
   288→function LiveClock() {
   289→  const [time, setTime] = React.useState("")
   290→
   291→  React.useEffect(() => {
   292→    const tick = () => {
   293→      setTime(
   294→        new Date().toLocaleTimeString("en-IN", {
   295→          hour: "2-digit",
   296→          minute: "2-digit",
   297→          second: "2-digit",
   298→          hour12: true,
   299→        })
   300→      )
   301→    }
   302→    tick()
   303→    const id = setInterval(tick, 1000)
   304→    return () => clearInterval(id)
   305→  }, [])
   306→
   307→  if (!time) return null
   308→  return (
   309→    <span className="hidden items-center gap-1.5 text-xs text-muted-foreground lg:flex">
   310→      <Clock className="h-3 w-3" />
   311→      {time}
   312→    </span>
   313→  )
   314→}

   532→export function CommandPalette() {
   533→  const { commandPaletteOpen, setCommandPaletteOpen, setActiveView } = useAppStore()
   534→  const [search, setSearch] = React.useState("")
   535→  const inputRef = React.useRef<HTMLInputElement>(null)
   536→
   537→  React.useEffect(() => {
   538→    const down = (e: KeyboardEvent) => {
   539→      if (e.key === "k" && (e.metaKey || e.ctrlKey)) {
   540→        e.preventDefault()
   541→        setCommandPaletteOpen(!commandPaletteOpen)
   542→      }
   543→      if (e.key === "Escape" && commandPaletteOpen) {
   544→        setCommandPaletteOpen(false)
   545→        setSearch("")
   546→      }
   547→    }
   548→    document.addEventListener("keydown", down)
   549→    return () => document.removeEventListener("keydown", down)
   550→  }, [commandPaletteOpen, setCommandPaletteOpen])
   551→
   552→  React.useEffect(() => {
   553→    if (commandPaletteOpen && inputRef.current) {
   554→      inputRef.current.focus()
   555→    }
   556→  }, [commandPaletteOpen])
   557→
   558→  const filtered = navItems.filter((item) =>
   559→    item.label.toLowerCase().includes(search.toLowerCase())
   560→  )
   561→
   562→  return (
   563→    <>
   564→      <div
   565→        className={cn(
   566→          "fixed inset-0 z-50 bg-black/40 backdrop-blur-sm transition-opacity duration-200",
   567→          commandPaletteOpen ? "opacity-100" : "pointer-events-none opacity-0"
   568→        )}
   569→        onClick={() => { setCommandPaletteOpen(false); setSearch("") }}
   570→      />
   571→      <div
   572→        className={cn(
   573→          "fixed left-1/2 top-[15%] z-50 w-full max-w-lg -translate-x-1/2 rounded-2xl border border-border/60 bg-background shadow-2xl shadow-black/20 transition-all duration-200",
   574→          commandPaletteOpen ? "scale-100 opacity-100" : "scale-95 opacity-0 pointer-events-none"
   575→        )}
   576→      >
   577→        <div className="flex items-center gap-2 border-b px-4">
   578→          <Search className="h-4 w-4 shrink-0 text-muted-foreground" />
   579→          <input
   580→            ref={inputRef}
   581→            placeholder="Search pages, warehouses, actions..."
   582→            className="flex-1 bg-transparent py-3.5 text-sm outline-none placeholder:text-muted-foreground"
   583→            value={search}
   584→            onChange={(e) => setSearch(e.target.value)}
   585→          />
   586→          <kbd className="pointer-events-none inline-flex h-5 select-none items-center rounded border bg-muted/80 px-1.5 font-mono text-[10px] font-medium text-muted-foreground">
   587→            ESC
   588→          </kbd>
   589→        </div>
   590→        <div className="max-h-72 overflow-y-auto p-2">
   591→          <div className="px-2 py-1.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   592→            {search ? "Results" : "Quick Navigation"}
   593→          </div>
   594→          {filtered.map((item) => {
   595→            const Icon = iconMap[item.icon]
   596→            const isActive = useAppStore.getState().activeView === item.id
   597→            return (
   598→              <button
   599→                key={item.id}
   600→                className={cn(
   601→                  "flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm text-left transition-colors",
   602→                  isActive ? "bg-blue-50 text-blue-700 dark:bg-blue-950 dark:text-blue-300" : "hover:bg-muted"
   603→                )}
   604→                onClick={() => {
   605→                  setActiveView(item.id)
   606→                  setCommandPaletteOpen(false)
   607→                  setSearch("")
   608→                }}
   609→              >
   610→                {Icon ? <Icon className="h-4 w-4 shrink-0" /> : <div className="h-4 w-4" />}
   611→                <span className="flex-1">{item.label}</span>
   612→                {isActive && <span className="text-[10px] text-muted-foreground">Current</span>}
   613→              </button>
   614→            )
   615→          })}
   616→          {filtered.length === 0 && (
   617→            <div className="px-3 py-8 text-center text-sm text-muted-foreground">
   618→              No results for &ldquo;{search}&rdquo;
   619→            </div>
   620→          )}
   621→        </div>
   622→        <div className="border-t px-4 py-2 flex items-center justify-between text-[10px] text-muted-foreground">
   623→          <span>Type to search</span>
   624→          <span className="flex items-center gap-1">
   625→            <kbd className="rounded border bg-muted px-1 py-0.5 font-mono text-[9px]">↑↓</kbd> navigate
   626→            <kbd className="rounded border bg-muted px-1 py-0.5 font-mono text-[9px] ml-2">↵</kbd> select
   627→            <kbd className="rounded border bg-muted px-1 py-0.5 font-mono text-[9px] ml-2">?</kbd> shortcuts
   628→          </span>
   629→        </div>
   630→      </div>
   631→    </>
   632→  )
   633→}
